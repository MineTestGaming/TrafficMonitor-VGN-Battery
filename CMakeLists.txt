cmake_minimum_required(VERSION 3.24)
project(TrafficMonitorBattery VERSION 0.1.2 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Use MSVC on Windows to match the TrafficMonitor C++ ABI")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8 /W4 /permissive- /EHsc)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
enable_testing()
add_executable(state_tests tests/state_tests.cpp)
target_include_directories(state_tests PRIVATE src)
add_test(NAME state COMMAND state_tests)
add_executable(protocol_tests tests/protocol_tests.cpp)
target_include_directories(protocol_tests PRIVATE src)
add_test(NAME hid_protocol COMMAND protocol_tests)
add_executable(BatteryCollector src/collector.cpp src/hid_backend.cpp)
target_link_libraries(BatteryCollector PRIVATE windowsapp setupapi hid)
add_executable(hid_inventory src/hid_inventory.cpp)
target_link_libraries(hid_inventory PRIVATE setupapi hid)
add_library(VgnVxeBattery SHARED src/plugin.cpp)
target_include_directories(VgnVxeBattery SYSTEM PRIVATE third_party)
target_link_libraries(VgnVxeBattery PRIVATE user32)
target_compile_options(VgnVxeBattery PRIVATE /wd4100)
add_executable(plugin_smoke tests/plugin_smoke.cpp)
target_include_directories(plugin_smoke PRIVATE third_party)
target_compile_options(plugin_smoke PRIVATE /wd4100)
target_link_options(plugin_smoke PRIVATE "/MANIFESTDEPENDENCY:type='win32' name='Microsoft.Windows.Common-Controls' version='6.0.0.0' processorArchitecture='*' publicKeyToken='6595b64144ccf1df' language='*'")
add_test(NAME plugin_abi COMMAND plugin_smoke "$<TARGET_FILE_DIR:BatteryCollector>/VgnVxeBattery.dll")
add_executable(sleeping_collector tests/sleeping_collector.cpp)
set_target_properties(sleeping_collector PROPERTIES OUTPUT_NAME BatteryCollector RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/timeout_fixture")
add_dependencies(sleeping_collector VgnVxeBattery)
add_custom_command(TARGET sleeping_collector POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:VgnVxeBattery>" "$<TARGET_FILE_DIR:sleeping_collector>/VgnVxeBattery.dll")
add_executable(helper_lifecycle tests/helper_lifecycle.cpp)
target_include_directories(helper_lifecycle SYSTEM PRIVATE third_party)
target_compile_options(helper_lifecycle PRIVATE /wd4100)
add_test(NAME helper_lifecycle COMMAND helper_lifecycle "$<TARGET_FILE_DIR:sleeping_collector>/VgnVxeBattery.dll")
set_tests_properties(helper_lifecycle PROPERTIES TIMEOUT 45)
add_executable(refresh_collector tests/refresh_collector.cpp)
set_target_properties(refresh_collector PROPERTIES OUTPUT_NAME BatteryCollector RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/refresh_fixture")
add_dependencies(refresh_collector VgnVxeBattery)
add_custom_command(TARGET refresh_collector POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:VgnVxeBattery>" "$<TARGET_FILE_DIR:refresh_collector>/VgnVxeBattery.dll")
add_executable(manual_refresh tests/manual_refresh.cpp)
target_include_directories(manual_refresh SYSTEM PRIVATE third_party)
target_compile_options(manual_refresh PRIVATE /wd4100)
add_test(NAME manual_refresh COMMAND manual_refresh "$<TARGET_FILE_DIR:refresh_collector>/VgnVxeBattery.dll")
set_tests_properties(manual_refresh PROPERTIES TIMEOUT 20)
